cmake_minimum_required(VERSION 3.22)
project(ShengBrightnessRestoreTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Test)
find_package(KF6Screen REQUIRED)
find_package(KF6I18n REQUIRED)
if(NOT EXISTS "${POWERDEVIL_SOURCE_DIR}/daemon/controllers/kwinbrightness.cpp")
    message(FATAL_ERROR "Set POWERDEVIL_SOURCE_DIR to the reviewed PowerDevil source")
endif()
set(controllers "${POWERDEVIL_SOURCE_DIR}/daemon/controllers")
add_executable(brightnessrestoretest brightnessrestoretest.cpp
    "${controllers}/kwinbrightness.cpp" "${controllers}/kwinbrightness.h"
    "${controllers}/displaybrightness.cpp" "${controllers}/displaybrightness.h")
target_include_directories(brightnessrestoretest PRIVATE "${controllers}")
target_link_libraries(brightnessrestoretest Qt6::Core Qt6::Test KF6::Screen KF6::I18n)
enable_testing()
add_test(NAME brightness_restore COMMAND brightnessrestoretest)
