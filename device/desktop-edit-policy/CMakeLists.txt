cmake_minimum_required(VERSION 3.25)
project(ShengDesktopEditPolicy LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
include(KDEInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick DBus)
find_package(KF6Config REQUIRED)
find_package(KF6CoreAddons REQUIRED)
find_package(KF6Kirigami REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(Plasma 6.7.4 EXACT REQUIRED)
find_package(PlasmaQuick 6.7.4 EXACT REQUIRED)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/policy.cpp" code_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/metadata.json" metadata_hash)
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/CMakeLists.txt" build_hash)
string(SHA256 policy_hash "${code_hash}${metadata_hash}${build_hash}")
string(SUBSTRING "${policy_hash}" 0 12 revision)
set(POLICY_ID "org.sheng.desktopeditpolicy.${revision}")
configure_file(metadata.json "${CMAKE_CURRENT_BINARY_DIR}/metadata.json" @ONLY)
configure_file(policy.cpp "${CMAKE_CURRENT_BINARY_DIR}/policy.cpp" COPYONLY)
add_library(${POLICY_ID} MODULE "${CMAKE_CURRENT_BINARY_DIR}/policy.cpp")
set_target_properties(${POLICY_ID} PROPERTIES PREFIX "")
target_link_libraries(${POLICY_ID} PRIVATE Plasma::Plasma Plasma::PlasmaQuick Qt6::Quick Qt6::Qml Qt6::DBus)
install(TARGETS ${POLICY_ID} DESTINATION ${KDE_INSTALL_PLUGINDIR}/plasma/applets)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/policy-id.txt" "${POLICY_ID}\n")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/metadata.json" DESTINATION "${KDE_INSTALL_DATADIR}/plasma/plasmoids/${POLICY_ID}")
install(DIRECTORY contents DESTINATION "${KDE_INSTALL_DATADIR}/plasma/plasmoids/${POLICY_ID}")
